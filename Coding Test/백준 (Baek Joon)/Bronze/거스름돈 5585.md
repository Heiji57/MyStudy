```python
def solve(n):
    money = 1000 - n
    res = 0
    coin_type = [500, 100, 50, 10, 5, 1]

    for coin in coin_type:
        res += (money // coin)
        money %= coin

    return res


if __name__ == '__main__':
    n = int(input())
    print(solve(n))
```
