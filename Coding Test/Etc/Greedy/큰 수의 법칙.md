### 반복문 사용
```python
import sys

def solve(M, K, array):
    array.sort(reverse=True)

    res = 0
    temp = 0
    for i in range(M):
        if temp >= K:
            res += array[1]
            temp = 0
        else:
            res += array[0]
            temp += 1

    return res


if __name__ == '__main__':
    N, M, K = map(int, sys.stdin.readline().split())
    array = list(map(int, sys.stdin.readline().split()))
    print(solve(M, K, array))
```
### 반복문 미사용
```python
import math
import sys

def solve(M, K, array):
    array.sort(reverse=True)

    k = 0
    if M % K == 0:
        k = math.trunc(M / K - 1)
    else:
        k = math.trunc(M / K)
    return array[0] * (M - k) + array[1] * k


if __name__ == '__main__':
    N, M, K = map(int, sys.stdin.readline().split())
    array = list(map(int, sys.stdin.readline().split()))
    print(solve(M, K, array))
```
