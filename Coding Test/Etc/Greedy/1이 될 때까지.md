```python
import sys

def solve(N, K):
    cnt = N
    res = 0
    while cnt != 1:
        if cnt % K == 0:
            cnt //= 4
            res += 1
        else:
            cnt -= 1
            res += 1

    return res

if __name__ == '__main__':
    N, K = map(int, sys.stdin.readline().split())

    print(solve(N, K))
```
