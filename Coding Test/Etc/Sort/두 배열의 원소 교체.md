def solve():
    N, M = map(int, input().split())

    arr_a = list(map(int, input().split()))
    arr_b = list(map(int, input().split()))

    arr_a.sort()
    arr_b.sort(reverse=True)
    for i in range(M):
        if arr_a[i] < arr_b[i]:
            arr_a[i], arr_b[i] = arr_b[i], arr_a[i]

    print(sum(arr_a))

if __name__ == '__main__':
    solve()
