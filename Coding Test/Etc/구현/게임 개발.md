```python
import sys


def solve():
    N, M = map(int, sys.stdin.readline().split())
    x, y, d = map(int, sys.stdin.readline().split())
    island = [list(map(int, sys.stdin.readline().split())) for _ in range(N)]

    dx = [-1, 0, 1, 0]
    dy = [0, 1, 0, -1]

    island[x][y] = 2
    res = 1
    count = 0

    while True:
        d = (d + 3) % 4
        count += 1

        nx = x + dx[d]
        ny = y + dy[d]

        if island[nx][ny] == 0:
            island[nx][ny] = 2
            x, y = nx, ny
            res += 1
            count = 0
            continue

        else:
            if count == 4:
                nx = x - dx[d]
                ny = y - dy[d]

                if island[nx][ny] == 1:
                    break

                else:
                    x, y = nx, ny
                    count = 0

    return res

if __name__ == '__main__':
    print(solve())
```
