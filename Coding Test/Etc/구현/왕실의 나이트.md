import sys


def solve():
    N = input()
    y = int(N[1])
    x = ord(N[0]) % 96

    nx = [-2, 2]
    second_ny = [-1, 1]
    ny = [-2, 2]
    second_nx = [-1, 1]

    res = 0

    for i in range(2):
        for j in range(2):
            if 1 <= x + nx[i] < 9 and 1 <= y + second_ny[j] < 9:
                res += 1
            if 1 <= y + ny[i] < 9 and 1 <= x + second_nx[j] < 9:
                res += 1

    return res

if __name__ == '__main__':
    print(solve())
