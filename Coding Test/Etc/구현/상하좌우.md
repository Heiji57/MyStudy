```python
import sys


def solve():
    N = int(input())
    control = map(str, sys.stdin.readline().split())
    leftRight = 1
    topBottom = 1
    for c in control:
        if c == 'R':
            if leftRight + 1 != N:
                leftRight += 1
        if c == 'L':
            if leftRight - 1 != 0:
                leftRight -= 1
        if c == 'D':
            if topBottom + 1 != N:
                topBottom += 1
        if c == 'U':
            if topBottom - 1 != 0:
                topBottom -= 1

    return topBottom, leftRight

if __name__ == '__main__':
    print(solve())
```
