def solve():
    res = 0
    n = int(input())
    for i in range(n+1):
        for j in range(60):
            for k in range(60):
                if '3' in (str(k) + str(j) + str(i)):
                    res += 1

    return res

if __name__ == '__main__':
    print(solve())
