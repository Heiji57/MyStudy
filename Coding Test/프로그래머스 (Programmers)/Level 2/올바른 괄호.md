from collections import deque

def solution(s):
    stack = deque(s)

    list = deque()
    for i in range(len(s)):
        m = stack.pop()
        if m == ')':
            list.append(m)
        if m == '(':
            if not list:
                return False
            list.popleft()
            
    if list:
        return False

    return True
