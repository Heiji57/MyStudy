### 내가 짠거
```python
from collections import deque

def solution(prices):
    dp = deque(prices)
    result = []
    for i in range(len(prices)):
        count = 0
        price = dp.popleft()
        for j in dp:
            print(f'{i} 루프 {j} 비교 {price} count: {count}')
            if i + 1 < price:
                count += 1
            if j >= price:
                count += 1
        result.append(count)

    return result
```
### AI가 짜준거
```python
from collections import deque

def solution(prices):
    dq = deque(prices)
    result = []

    while dq:
        price = dq.popleft()
        count = 0

        for next_price in dq:
            count += 1
            if next_price < price:
                break
        result.append(count)
    return result
```
