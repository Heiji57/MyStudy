```python
from collections import deque

def solution(progresses, speeds):
    result = []
    while progresses:
        cnt = 0
        while progresses and progresses[0] >= 100:
            cnt += 1
            progresses.pop(0)
            speeds.pop(0)
        progresses = [progresses[i]+speeds[i] for i in range(len(progresses))]

        if cnt > 0:
            result.append(cnt)

    return result
```
